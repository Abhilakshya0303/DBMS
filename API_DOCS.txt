# 2. AUTHENTICATION
# ══════════════════════════════════════════════════════════════

## 2A. Register
POST http://localhost:5000/api/auth/register
Content-Type: application/json

BODY:
{
  "full_name": "John Doe",
  "email":     "john@thapar.edu",
  "password":  "password123",
  "role":      "Requester",
  "dept_id":   1
}

RESPONSE (201):
{
  "success": true,
  "message": "User registered successfully.",
  "user_id": 6
}

# 4. REQUESTS
# ══════════════════════════════════════════════════════════════

## 4A. Submit Room Request
POST http://localhost:5000/api/requests
Authorization: Bearer <REQUESTER_TOKEN>
Content-Type: application/json

BODY:
{
  "event_title":  "IEEE Workshop on Machine Learning",
  "event_type":   "Workshop",
  "dept_id":      1,
  "room_id":      5,
  "request_date": "2025-06-15",
  "slot_id":      6
}

RESPONSE SUCCESS (201):
{
  "success":    true,
  "message":    "Request submitted successfully.",
  "request_id": 3
}

RESPONSE CONFLICT (409):
{
  "success": false,
  "message": "Room is already booked for the selected date and slot."
}

## 4B. Get My Requests
GET http://localhost:5000/api/requests/mine
Authorization: Bearer <REQUESTER_TOKEN>

RESPONSE (200):
{
  "success": true,
  "data": [
    {
      "request_id":   3,
      "event_title":  "IEEE Workshop on Machine Learning",
      "event_type":   "Workshop",
      "room_code":    "SH-101",
      "building":     "Main Block",
      "slot_label":   "Slot 6 (13:00–14:00)",
      "request_date": "2025-06-15",
      "status":       "Pending",
      "decided_by_name": null,
      "remarks":      null,
      "created_at":   "2025-06-01T10:00:00.000Z"
    }
  ]
}

## 4C. Get All Requests (Admin/Staff)
GET http://localhost:5000/api/requests
Authorization: Bearer <ADMIN_TOKEN>

QUERY PARAMS (optional):
  status=Pending   (Pending | Approved | Rejected)

## 4D. Approve / Reject Request
PATCH http://localhost:5000/api/requests/3/decide
Authorization: Bearer <ADMIN_TOKEN>
Content-Type: application/json

BODY (Approve):
{
  "decision": "Approved",
  "remarks":  "Approved for IEEE workshop. Please ensure the hall is cleaned after use."
}

BODY (Reject):
{
  "decision": "Rejected",
  "remarks":  "Hall is reserved for university function on this date."
}

RESPONSE SUCCESS (200):
{ "success": true, "message": "Request Approved successfully." }

RESPONSE (already decided):
{ "success": false, "message": "Request has already been decided." }

# 6. ERROR RESPONSES
# ══════════════════════════════════════════════════════════════

401 Unauthorized:
{ "success": false, "message": "Access denied. No token provided." }

403 Forbidden:
{ "success": false, "message": "Access denied. Required role(s): Admin, Staff." }

404 Not Found:
{ "success": false, "message": "Route GET /api/unknown not found." }

409 Conflict (Double Booking via DB Trigger):
{ "success": false, "message": "CONFLICT: Room is already allocated for this date and time slot." }

422 Validation Error:
{
  "success": false,
  "errors": [
    { "msg": "Valid email is required.", "path": "email" }
  ]
}
